💻 مثال: AI بيكتب experiment

# AI generates this experiment automatically
from src.experiments.registry import ExperimentRegistry
import numpy as np

@ExperimentRegistry.register(
    name="bitcoin_key_analysis_v2",
    description="Analyze key locality with new features",
    tags=["bitcoin", "crypto", "nightly"]
)
def bitcoin_key_analysis_v2(
    sample_size: int = 100_000,
    feature_dims: int = 24
) -> dict:
    # AI-generated experiment logic
    keys = np.random.randint(0, 2**256, size=sample_size)
    features = extract_features(keys, dims=feature_dims)
    return {
        "sample_size": sample_size,
        "locality_score": compute_locality(features),
        "entropy": compute_entropy(features),
    }

# Run: project-x run --name bitcoin_key_analysis_v2
# Or via API: POST /api/v1/experiments/run
